function Add-House {
  param(
    [Parameter(Mandatory=$true)]
    [string]$HouseSlug,
    [Parameter(Mandatory=$true)]
    [string]$AssetsRoot
  )

  $HouseSlug = $HouseSlug.ToLower()
  $paths = @(
    "portraits\elites\house_$HouseSlug",
    "portraits\mutants\loyalist\house_$HouseSlug",
    "portraits\mutants\ronin\house_$HouseSlug"
  )

  foreach ($p in $paths) {
    New-Item -ItemType Directory -Force -Path (Join-Path $AssetsRoot $p) | Out-Null
  }

  Write-Host "Added House folders for: $HouseSlug"
}

# Example:
# Add-House -HouseSlug "house_newname" -AssetsRoot "C:\path\to\your\assets"
